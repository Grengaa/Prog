cmake_minimum_required(VERSION 3.21)
project(ArduinoPlotter VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets SerialPort)
qt_add_executable(ArduinoPlotter src/main.cpp src/mainwindow.cpp)
target_link_libraries(ArduinoPlotter PRIVATE Qt6::Widgets Qt6::SerialPort)
set_target_properties(ArduinoPlotter PROPERTIES WIN32_EXECUTABLE TRUE)
include(CTest)
if(BUILD_TESTING)
  add_executable(parser_test tests/parser_test.cpp)
  add_test(NAME parser_test COMMAND parser_test)
  add_executable(serialcsvbuffer_test tests/serialcsvbuffer_test.cpp)
  target_link_libraries(serialcsvbuffer_test PRIVATE Qt6::Core)
  add_test(NAME serialcsvbuffer_test COMMAND serialcsvbuffer_test)
endif()
